aiortc>=1.14,<2                     # WHIP publisher — webrtc_pub.py
av>=16,<17                          # PyAV H.264 encode for aiortc
# Pin <2: ros:jazzy-ros-base ships numpy 1.x and rclpy / the opencv 4.9 wheel are
# built against the numpy 1.x ABI — numpy 2.x would break `import cv2` / rclpy.
numpy>=1.24,<2                      # depth-map math (stereo.py); also an opencv dependency
